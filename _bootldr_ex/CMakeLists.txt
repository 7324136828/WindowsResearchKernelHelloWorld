cmake_minimum_required(VERSION 3.20)
project(REACTOS C CXX)
include(CMakeDependentOption)

# This directory is a portable ReactOS source snapshot, with a focused build.
set(ARCH "i386" CACHE STRING "Target architecture: i386 or amd64")
string(TOLOWER "${ARCH}" ARCH)
if(NOT ARCH MATCHES "^(i386|amd64)$")
    message(FATAL_ERROR "The focused WRK component build supports i386 and amd64")
endif()
if(ARCH STREQUAL "i386")
    set(WINARCH x86)
else()
    set(WINARCH ${ARCH})
endif()
set(CMAKE_INCLUDE_CURRENT_DIR ON)
set(CMAKE_INCLUDE_DIRECTORIES_PROJECT_BEFORE ON)
set(CMAKE_SHARED_LIBRARY_PREFIX "")
set(CMAKE_SHARED_MODULE_PREFIX "")
set(CMAKE_SKIP_PREPROCESSED_SOURCE_RULES TRUE)
set(CMAKE_SKIP_ASSEMBLY_SOURCE_RULES TRUE)
set(CMAKE_POSITION_INDEPENDENT_CODE OFF)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build configuration" FORCE)
endif()
# The focused build uses DWARF rather than rsym; no dbghelp DLL is required.
set(NO_ROSSYM TRUE CACHE BOOL "Use DWARF symbols in the focused build")
if(NOT NO_ROSSYM)
    message(FATAL_ERROR "The focused build requires NO_ROSSYM=ON")
endif()
add_compile_definitions(__REACTOS__ __RELFILE__=__FILE__)
include(sdk/include/reactos/version.cmake)
include(sdk/cmake/config.cmake)
include(sdk/cmake/compilerflags.cmake)
include(sdk/cmake/set_wine_module.cmake)
include(cmake/data.cmake)

if(CMAKE_CROSSCOMPILING)
    if(SARCH STREQUAL "pc98" OR SARCH STREQUAL "xbox")
        message(FATAL_ERROR "The focused WRK build uses the PC BIOS platform")
    endif()
    include(cmake/host-tools.cmake)
    setup_host_tools()
    include(cmake/reactos-target.cmake)
    # The modules retain upstream media-registration calls. This bundle stages
    # selected outputs directly instead of assembling a ReactOS installation CD.
    function(add_cd_file)
    endfunction()
    include(cmake/components.cmake)
    bootldr_ex_add_data(native-mkhive native-txt2nls native-utf16le)
else()
    option(BOOTLDR_EX_BUILD_CODEGEN "Also build target-code and IDL generators" OFF)
    if(NOT DEFINED TOOLS_FOLDER)
        set(TOOLS_FOLDER "${CMAKE_BINARY_DIR}/bin")
    endif()
    add_compile_definitions(TARGET_${ARCH})
    if(MSVC)
        add_compile_definitions(_CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_DEPRECATE HAVE_IO_H=1)
        add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/EHsc>" /wd4267)
    endif()
    function(add_host_tool tool)
        add_executable(${tool} ${ARGN})
        set_target_properties(${tool} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${TOOLS_FOLDER}")
    endfunction()
    function(add_host_module module)
        add_library(${module} MODULE ${ARGN})
        set_target_properties(${module} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${TOOLS_FOLDER}")
    endfunction()
    add_subdirectory(sdk/include/host)
    add_subdirectory(sdk/lib/cmlib)
    add_subdirectory(sdk/lib/inflib)
    add_subdirectory(sdk/tools/unicode)
    add_subdirectory(sdk/tools/mkhive)
    add_subdirectory(sdk/tools/txt2nls)
    add_host_tool(utf16le sdk/tools/utf16le/utf16le.cpp)
    if(BOOTLDR_EX_BUILD_CODEGEN)
        find_package(BISON REQUIRED)
        find_package(FLEX REQUIRED)
        add_subdirectory(sdk/tools/hpp)
        add_subdirectory(sdk/tools/asmpp)
        add_subdirectory(sdk/tools/wpp)
        add_subdirectory(sdk/tools/widl)
        add_host_tool(bin2c sdk/tools/bin2c.c)
        add_host_tool(geninc sdk/tools/geninc/geninc.c)
        add_host_tool(obj2bin sdk/tools/obj2bin/obj2bin.c)
        target_link_libraries(obj2bin PRIVATE host_includes)
        add_host_tool(spec2def sdk/tools/spec2def/spec2def.c)
        if(NOT MSVC)
            add_host_tool(pefixup sdk/tools/pefixup.c)
            target_link_libraries(pefixup PRIVATE host_includes)
            if(ARCH STREQUAL "amd64")
                target_compile_definitions(pefixup PRIVATE _TARGET_PE64)
            endif()
        endif()
        if(ARCH STREQUAL "amd64" AND TARGET_COMPILER_ID STREQUAL "GNU")
            add_subdirectory(sdk/tools/gcc_plugin_seh)
        endif()
    endif()
    bootldr_ex_add_data(mkhive txt2nls utf16le)
endif()
