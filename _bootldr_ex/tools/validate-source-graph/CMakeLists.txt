function(validation_path path base output)
    get_filename_component(resolved "${path}" ABSOLUTE BASE_DIR "${base}")
    file(TO_CMAKE_PATH "${resolved}" resolved)
    file(RELATIVE_PATH source_relative "${REACTOS_SOURCE_DIR}" "${resolved}")
    file(RELATIVE_PATH output_relative "${CMAKE_BINARY_DIR}" "${resolved}")
    if((source_relative MATCHES "^\\.\\./" OR IS_ABSOLUTE "${source_relative}") AND
       (output_relative MATCHES "^\\.\\./" OR IS_ABSOLUTE "${output_relative}"))
        message(FATAL_ERROR "Path escapes the portable bundle and validation output: ${resolved}")
    endif()
    set(${output} "${resolved}" PARENT_SCOPE)
endfunction()

function(check_existing input)
    validation_path("${input}" "${CMAKE_CURRENT_SOURCE_DIR}" path)
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Missing source or helper input: ${path}")
    endif()
    set_property(GLOBAL APPEND PROPERTY VALIDATION_INPUTS "${path}")
endfunction()

function(validation_placeholder path)
    validation_path("${path}" "${CMAKE_CURRENT_BINARY_DIR}" path)
    file(RELATIVE_PATH relative "${CMAKE_BINARY_DIR}" "${path}")
    if(relative MATCHES "^\\.\\./" OR IS_ABSOLUTE "${relative}")
        message(FATAL_ERROR "A generated placeholder must be in the output directory: ${path}")
    endif()
    get_filename_component(parent "${path}" DIRECTORY)
    file(MAKE_DIRECTORY "${parent}")
    if(NOT EXISTS "${path}")
        file(WRITE "${path}" "/* Configure-only placeholder; not a generated ReactOS artifact. */\n")
    endif()
    set_property(GLOBAL APPEND PROPERTY VALIDATION_GENERATED "${path}")
endfunction()

function(spec2def filename spec)
    check_existing("${spec}")
    get_filename_component(stem "${filename}" NAME_WE)
    validation_placeholder("${CMAKE_CURRENT_BINARY_DIR}/${stem}.def")
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/${stem}.def" PROPERTIES HEADER_FILE_ONLY TRUE)

        add_library(${target} INTERFACE)

    endif()
endfunction()

    endforeach()
endfunction()

function(validate_directory directory)
    get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
    foreach(target IN LISTS targets)
        get_target_property(kind "${target}" TYPE)
        get_target_property(source_dir "${target}" SOURCE_DIR)
        set_property(GLOBAL APPEND PROPERTY VALIDATION_TARGETS "${target}:${kind}")
        # The harness is configure-only; its default build has no source targets.
        set_target_properties("${target}" PROPERTIES EXCLUDE_FROM_ALL TRUE)
        foreach(property IN ITEMS INCLUDE_DIRECTORIES INTERFACE_INCLUDE_DIRECTORIES)
            get_target_property(includes "${target}" "${property}")
            if(NOT includes OR includes MATCHES "-NOTFOUND$")
                continue()
            endif()
            foreach(include IN LISTS includes)
                if(include MATCHES "^\\$<BUILD_INTERFACE:(.*)>$")
                    set(include "${CMAKE_MATCH_1}")
                elseif(include MATCHES "\\$<")
                    message(FATAL_ERROR "Cannot validate target ${target} include expression: ${include}")
                endif()
                validation_path("${include}" "${source_dir}" path)
                if(NOT IS_DIRECTORY "${path}")
                    # Upstream math.cmake has an unused, stale relative include.
                    # Keep copied source hashes intact; allow only this exact path
                    # while checking the real sibling math/libm_sse2 directory.
                    if(path STREQUAL "${REACTOS_SOURCE_DIR}/sdk/lib/crt/libm_sse2")
                        check_existing("${REACTOS_SOURCE_DIR}/sdk/lib/crt/math/libm_sse2")
                        set_property(GLOBAL APPEND PROPERTY VALIDATION_UPSTREAM_EXCEPTIONS "${path}:upstream-unused-include")
                        continue()
                    endif()
                    message(FATAL_ERROR "Target ${target} references missing include directory ${path}")
                endif()
                set_property(GLOBAL APPEND PROPERTY VALIDATION_INCLUDES "${path}")
            endforeach()
        endforeach()
        foreach(property IN ITEMS LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
            get_target_property(links "${target}" "${property}")
            if(NOT links OR links MATCHES "-NOTFOUND$")
                continue()
            endif()
            foreach(link IN LISTS links)
                if(link MATCHES "^\\$<LINK_ONLY:(.*)>$")
                    set(link "${CMAKE_MATCH_1}")
                elseif(link MATCHES "\\$<")
                    continue()
                endif()
                if(NOT TARGET "${link}" AND NOT link MATCHES "^[-/]")
                    message(FATAL_ERROR "Target ${target} links to an unavailable dependency: ${link}")
                endif()
                set_property(GLOBAL APPEND PROPERTY VALIDATION_LINKS "${target}->${link}")
            endforeach()
        endforeach()
        if(NOT kind STREQUAL "UTILITY" AND NOT kind STREQUAL "INTERFACE_LIBRARY")
            set_target_properties("${target}" PROPERTIES LINKER_LANGUAGE C)
        endif()
        get_target_property(sources "${target}" SOURCES)
        if(NOT sources OR sources MATCHES "-NOTFOUND$")
            continue()
        endif()
        foreach(source IN LISTS sources)
            if(source MATCHES "^\\$<TARGET_OBJECTS:([^>]+)>$")
                if(NOT TARGET "${CMAKE_MATCH_1}")
                    message(FATAL_ERROR "Target ${target} references unavailable object target ${CMAKE_MATCH_1}")
                endif()
                continue()
            elseif(source MATCHES "\\$<")
                message(FATAL_ERROR "Cannot validate target ${target} source expression: ${source}")
            endif()
            validation_path("${source}" "${source_dir}" path)
            get_source_file_property(generated "${path}" DIRECTORY "${directory}" GENERATED)
            if(generated)
                validation_placeholder("${path}")
            elseif(NOT EXISTS "${path}")
                message(FATAL_ERROR "Target ${target} references missing source ${path}")
            else()
                set_property(GLOBAL APPEND PROPERTY VALIDATION_INPUTS "${path}")
            endif()
            if(path MATCHES "\\.(rc|def|s|S)$")
                set_source_files_properties("${path}" DIRECTORY "${directory}" PROPERTIES HEADER_FILE_ONLY TRUE)
            endif()
        endforeach()
    endforeach()
    get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
    foreach(child IN LISTS children)
        validate_directory("${child}")
    endforeach()
endfunction()
validate_directory("${CMAKE_CURRENT_SOURCE_DIR}")

foreach(category IN ITEMS INPUTS INCLUDES GENERATED TARGETS LINKS STUBS UPSTREAM_EXCEPTIONS)
    get_property(values GLOBAL PROPERTY VALIDATION_${category})
    list(REMOVE_DUPLICATES values)
    list(SORT values)
    list(LENGTH values count)
    string(JOIN "\n" value_text ${values})
    string(APPEND report "\n${category}=${count}\n${value_text}\n")
endforeach()
